cmake_minimum_required(VERSION 3.24)
project(FreeFrameGen VERSION 0.3.0 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows only")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS shaders/interpolate.hlsl)
file(READ shaders/interpolate.hlsl FFG_SHADER_SOURCE)
configure_file(cmake/ffg_shader.hpp.in generated/ffg_shader.hpp @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS shaders/capture.hlsl)
file(READ shaders/capture.hlsl CAPTURE_SHADER_SOURCE)
configure_file(cmake/capture_shader.hpp.in generated/capture_shader.hpp @ONLY)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS assets/ffg.ico cmake/ffg_icon.rc.in)
file(TO_CMAKE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/assets/ffg.ico" FFG_ICON_FILE)
configure_file(cmake/ffg_icon.rc.in generated/ffg_icon.rc @ONLY)
set(FFG_ICON_RC "${CMAKE_CURRENT_BINARY_DIR}/generated/ffg_icon.rc")
add_library(FreeFrameGen SHARED src/ffg.cpp)
target_compile_definitions(FreeFrameGen PRIVATE FFG_BUILD)
target_include_directories(FreeFrameGen PUBLIC include PRIVATE support ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(FreeFrameGen PRIVATE d3d12 dxgi d3dcompiler)
add_executable(ffg_demo src/demo.cpp ${FFG_ICON_RC})
target_include_directories(ffg_demo PRIVATE support)
target_link_libraries(ffg_demo PRIVATE FreeFrameGen d3d12 dxgi d3dcompiler)
add_executable(ffg_capture src/capture.cpp ${FFG_ICON_RC})
target_include_directories(ffg_capture PRIVATE support ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(ffg_capture PRIVATE d3d11 d3d12 dxgi d3dcompiler)
add_executable(ffg_player WIN32 src/player_gui.cpp ${FFG_ICON_RC})
target_link_libraries(ffg_player PRIVATE comctl32 shell32 user32)
add_executable(ffg_flow_test tests/color_flow.cpp ${FFG_ICON_RC})
target_include_directories(ffg_flow_test PRIVATE support ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(ffg_flow_test PRIVATE d3d12 dxgi d3dcompiler)
foreach(t FreeFrameGen ffg_demo ffg_capture ffg_player ffg_flow_test)
  target_compile_features(${t} PRIVATE cxx_std_17)
  target_compile_options(${t} PRIVATE /W4 /WX /utf-8 /permissive-)
endforeach()
enable_testing()
add_test(NAME ffg_warp COMMAND ffg_demo --headless --warp --debug)
add_test(NAME color_flow_warp COMMAND ffg_flow_test --warp --debug)
add_test(NAME color_flow_stress_warp COMMAND ffg_flow_test --warp --stress --debug)
add_test(NAME capture_help COMMAND ffg_capture --help)
add_test(NAME player_help COMMAND ffg_player --help)
add_test(NAME capture_self_test COMMAND ffg_capture --self-test)
add_test(NAME capture_reject_invalid_width COMMAND ffg_capture --width 319)
set_tests_properties(capture_reject_invalid_width PROPERTIES WILL_FAIL TRUE)
add_executable(ffg_c_header tests/ffg_c.c)
target_include_directories(ffg_c_header PRIVATE include)
add_test(NAME c_header COMMAND ffg_c_header)
add_executable(fgds_vk_c_header tests/fgds_vk_c.c)
target_include_directories(fgds_vk_c_header PRIVATE include)
add_test(NAME fgds_vk_c_header COMMAND fgds_vk_c_header)
install(TARGETS FreeFrameGen ffg_demo ffg_capture ffg_player ffg_flow_test RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY docs/ DESTINATION docs)
install(DIRECTORY tools/ DESTINATION tools)
install(FILES assets/ffg.ico DESTINATION assets)
install(FILES README.md product.txt DESTINATION .)

# The Vulkan runtime has an independent SDK/glslang toolchain requirement.
# Keep it opt-in so the normal D3D12/capture build remains usable on machines
# without the Vulkan SDK, while allowing one top-level configure for native
# Vulkan integrations when the dependencies are available.
option(FFG_BUILD_VULKAN "Build the optional Vulkan native runtime" OFF)
if(FFG_BUILD_VULKAN)
  add_subdirectory(vulkan)
endif()
