cmake_minimum_required(VERSION 3.24)
project(FreeFrameGen VERSION 0.1.0 LANGUAGES C CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows only")
endif()
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS shaders/interpolate.hlsl)
file(READ shaders/interpolate.hlsl FFG_SHADER_SOURCE)
configure_file(cmake/ffg_shader.hpp.in generated/ffg_shader.hpp @ONLY)
add_library(FreeFrameGen SHARED src/ffg.cpp)
target_compile_definitions(FreeFrameGen PRIVATE FFG_BUILD)
target_include_directories(FreeFrameGen PUBLIC include PRIVATE support ${CMAKE_CURRENT_BINARY_DIR}/generated)
target_link_libraries(FreeFrameGen PRIVATE d3d12 dxgi d3dcompiler)
add_executable(ffg_demo src/demo.cpp)
target_include_directories(ffg_demo PRIVATE support)
target_link_libraries(ffg_demo PRIVATE FreeFrameGen d3d12 dxgi d3dcompiler)
foreach(t FreeFrameGen ffg_demo)
  target_compile_features(${t} PRIVATE cxx_std_17)
  target_compile_options(${t} PRIVATE /W4 /WX /utf-8 /permissive-)
endforeach()
enable_testing()
add_test(NAME ffg_warp COMMAND ffg_demo --headless --warp --debug)
add_executable(ffg_c_header tests/ffg_c.c)
target_include_directories(ffg_c_header PRIVATE include)
add_test(NAME c_header COMMAND ffg_c_header)
install(TARGETS FreeFrameGen ffg_demo RUNTIME DESTINATION bin ARCHIVE DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(DIRECTORY docs/ DESTINATION docs)
install(DIRECTORY tools/ DESTINATION tools)
install(FILES README.md product.txt DESTINATION .)
